cmake_minimum_required(VERSION 3.16)
project(personal_assistant_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(model_runtime STATIC model_runtime.cpp)
add_library(model_head_runtime STATIC head_runtime.cpp)
foreach(target model_runtime model_head_runtime)
    target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

add_executable(runtime_self_test self_test.cpp)
target_link_libraries(runtime_self_test PRIVATE model_runtime)
add_executable(head_self_test head_self_test.cpp)
target_link_libraries(head_self_test PRIVATE model_head_runtime)
enable_testing()
add_test(NAME runtime_self_test COMMAND runtime_self_test)
add_test(NAME head_self_test COMMAND head_self_test)

set(MODEL_DATA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../model_data" CACHE PATH
    "Directory containing generated model_data.h and model_head.h")
if(EXISTS "${MODEL_DATA_DIR}/model_data.h")
    add_executable(model_runner host_runner.cpp)
    target_include_directories(model_runner PRIVATE "${MODEL_DATA_DIR}")
    target_link_libraries(model_runner PRIVATE model_runtime)
else()
    message(STATUS "model_data.h absent: full runner will not be built.")
endif()
if(EXISTS "${MODEL_DATA_DIR}/model_head.h")
    add_executable(head_runner head_runner.cpp)
    target_include_directories(head_runner PRIVATE "${MODEL_DATA_DIR}")
    target_link_libraries(head_runner PRIVATE model_head_runtime)
else()
    message(STATUS "model_head.h absent: head runner will not be built.")
endif()

add_library(task_state STATIC task_state.cpp)
target_include_directories(task_state PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(task_state_self_test task_state_self_test.cpp)
target_link_libraries(task_state_self_test PRIVATE task_state)
add_test(NAME task_state_self_test COMMAND task_state_self_test)

if(EXISTS "${MODEL_DATA_DIR}/model_data.h")
    add_executable(classifier_service classifier_service.cpp)
    target_include_directories(classifier_service PRIVATE "${MODEL_DATA_DIR}")
    target_link_libraries(classifier_service PRIVATE model_runtime model_head_runtime task_state)
endif()
